Tamagocki style cat game

Menus:
  Cat menu: 
    direction buttons can select where the cat looks
    btn_OK = Main menu
  Main menu:
    btn_LEFT = happiness
    btn_UP = hunger
    btn_DOWN = energy
    btn_RIGHT = health
    btn_OK = Cat menu / Select

Activities:
  Gams:
    laser pointer: click the dot -> +2 happy on win
    purr: button mash -> +1 happy on win
  Feed:
    fish -> +2 hunger
    chicken -> +4 hunger
    beetle -> +1 hunger +1 happy
  Sleep:
    4 hours -> +8 energy
  Medicine:
    vaccine -> +4 health

Decays to zero:
  Happiness: 12 hours
  Hunger: 6 hours
  Energy: 8 hours
  Health: 2-5 days randomised
